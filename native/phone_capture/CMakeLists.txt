cmake_minimum_required(VERSION 3.16)
project(phonecapture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT EXISTS "${JSON_INCLUDE}/nlohmann/json.hpp")
  message(FATAL_ERROR "Set JSON_INCLUDE to the pinned nlohmann include directory")
endif()
add_library(phonecapture SHARED napi.cpp ../screen_capture.cpp)
target_include_directories(phonecapture PRIVATE .. "${JSON_INCLUDE}")
target_compile_definitions(phonecapture PRIVATE GAMEVOICE_CAPTURE_ONLY=1)
target_compile_options(phonecapture PRIVATE -fvisibility=hidden -fstack-protector-strong)
target_link_options(phonecapture PRIVATE -Wl,-z,relro,-z,now -Wl,--no-undefined)
target_link_libraries(phonecapture PRIVATE ace_napi.z z dl m)
